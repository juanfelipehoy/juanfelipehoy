<script>

const canvas=document.getElementById("game");
const ctx=canvas.getContext("2d");

const scoreEl=document.getElementById("score");
const highEl=document.getElementById("high");
const overlay=document.getElementById("overlay");

let score=0;
let high=Number(localStorage.getItem("juanHighScore")||0);

highEl.textContent=String(high).padStart(6,"0");

let running=false;

const player={
    x:0,
    y:0,
    r:10,
    speed:3.2,
    dx:0,
    dy:0
};

const dots=[];

function resize(){

    const rect=canvas.getBoundingClientRect();

    canvas.width=rect.width*devicePixelRatio;
    canvas.height=rect.height*devicePixelRatio;

    ctx.scale(devicePixelRatio,devicePixelRatio);

}

window.addEventListener("resize",resize);

resize();

function createDots(){

    dots.length=0;

    const width=canvas.clientWidth;
    const height=canvas.clientHeight;

    for(let x=35;x<width-30;x+=34){

        for(let y=35;y<height-30;y+=34){

            dots.push({
                x,
                y,
                active:true
            });

        }

    }

}

function startGame(){

    score=0;

    scoreEl.textContent="000000";

    player.x=canvas.clientWidth/2;
    player.y=canvas.clientHeight/2;

    player.dx=0;
    player.dy=0;

    createDots();

    overlay.style.display="none";

    running=true;

    requestAnimationFrame(loop);

}

function drawBackground(){

    const width=canvas.clientWidth;
    const height=canvas.clientHeight;

    ctx.fillStyle="#02030a";
    ctx.fillRect(0,0,width,height);

    ctx.strokeStyle="rgba(0,240,255,.08)";
    ctx.lineWidth=2;

    ctx.strokeRect(
        20,
        20,
        width-40,
        height-40
    );

}

function drawDots(){

    for(const dot of dots){

        if(!dot.active) continue;

        ctx.beginPath();

        ctx.arc(
            dot.x,
            dot.y,
            2,
            0,
            Math.PI*2
        );

        ctx.fillStyle="#B892FF";

        ctx.shadowBlur=8;
        ctx.shadowColor="#B892FF";

        ctx.fill();

        ctx.shadowBlur=0;

    }

}

function drawPlayer(){

    ctx.save();

    ctx.translate(
        player.x,
        player.y
    );

    let angle=0;

    if(player.dx<0) angle=Math.PI;
    if(player.dy<0) angle=-Math.PI/2;
    if(player.dy>0) angle=Math.PI/2;

    ctx.rotate(angle);

    ctx.beginPath();

    ctx.moveTo(0,0);

    ctx.arc(
        0,
        0,
        player.r,
        .25*Math.PI,
        1.75*Math.PI
    );

    ctx.closePath();

    ctx.fillStyle="#FFD93D";

    ctx.shadowBlur=20;
    ctx.shadowColor="#FFD93D";

    ctx.fill();

    ctx.restore();

    ctx.shadowBlur=0;

}

function update(){

    player.x+=player.dx;
    player.y+=player.dy;

    const width=canvas.clientWidth;
    const height=canvas.clientHeight;

    if(player.x<25) player.x=width-25;
    if(player.x>width-25) player.x=25;

    if(player.y<25) player.y=height-25;
    if(player.y>height-25) player.y=25;

    for(const dot of dots){

        if(!dot.active) continue;

        const dx=player.x-dot.x;
        const dy=player.y-dot.y;

        const distance=Math.sqrt(dx*dx+dy*dy);

        if(distance<12){

            dot.active=false;

            score+=10;

            scoreEl.textContent=
                String(score).padStart(6,"0");

            if(score>high){

                high=score;

                highEl.textContent=
                    String(high).padStart(6,"0");

                localStorage.setItem(
                    "juanHighScore",
                    high
                );

            }

        }

    }

}

function draw(){

    drawBackground();

    drawDots();

    drawPlayer();

}

function loop(){

    if(!running) return;

    update();

    draw();

    requestAnimationFrame(loop);

}

document.addEventListener("keydown",e=>{

    if(
        e.key==="ArrowLeft" ||
        e.key.toLowerCase()==="a"
    ){

        player.dx=-player.speed;
        player.dy=0;

    }

    if(
        e.key==="ArrowRight" ||
        e.key.toLowerCase()==="d"
    ){

        player.dx=player.speed;
        player.dy=0;

    }

    if(
        e.key==="ArrowUp" ||
        e.key.toLowerCase()==="w"
    ){

        player.dy=-player.speed;
        player.dx=0;

    }

    if(
        e.key==="ArrowDown" ||
        e.key.toLowerCase()==="s"
    ){

        player.dy=player.speed;
        player.dx=0;

    }

});
